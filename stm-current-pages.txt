function formatStructureType(value: string) {
  const labels: Record<string, string> = {
    single_elimination: "Single Elimination",
    round_robin: "Round Robin",
    round_robin_bracket: "Round Robin + Bracket",
    round_robin_bracket_consolation: "Round Robin + Bracket + Consolation",
  }

  return labels[value] || value
}

function formatPlayMode(value: string) {
  const labels: Record<string, string> = {
    singles: "Singles",
    doubles: "Doubles",
    individual_doubles: "Individual Doubles",
  }

  return labels[value] || value
}

import { listCompetitionEntries } from "@/modules/competition-entries/repositories/competition-entry.repository"

import { getCompetitionAction } from "@/modules/competitions/actions/getCompetition"

type CompetitionDetailPageProps = {
  params: Promise<{
    id: string
  }>
  searchParams?: Promise<{
    section?: string
  }>
}

const sections = [
  {
    key: "configuration",
    label: "Configuration",
  },
  {
    key: "entries",
    label: "Entries",
  },
  {
    key: "generate",
    label: "Generate",
  },
  {
    key: "play",
    label: "Play",
  },
  {
    key: "reports",
    label: "Reports",
  },
]

function formatStructureType(value: string) {
  const labels: Record<string, string> = {
    single_elimination: "Single Elimination",
    round_robin: "Round Robin",
    round_robin_bracket:
      "Round Robin + Bracket",
    round_robin_bracket_consolation:
      "Round Robin + Bracket + Consolation",
  }

  return labels[value] || value
}

function formatPlayMode(value: string) {
  const labels: Record<string, string> = {
    singles: "Singles",
    doubles: "Doubles",
    individual_doubles:
      "Individual Doubles",
  }

  return labels[value] || value
}

export default async function CompetitionDetailPage({
  params,
  searchParams,
}: CompetitionDetailPageProps) {
  const member = await getCurrentMember()

  if (!member) {
    redirect("/login?error=no_active_member")
  }

  const { id } = await params
  const query = await searchParams

  const validSection =
    sections.find(
      (section) => section.key === query?.section
    )?.key ?? "configuration"

  const competition =
    await getCompetitionAction(id)

  if (!competition) {
    notFound()
  }

  const entries =
    validSection === "entries"
      ? await listCompetitionEntries(
          competition.id
        )
      : []

  const locked = [
    "generated",
    "running",
    "completed",
    "archived",
  ].includes(competition.status)

  return (
    <main className="min-h-screen bg-slate-50 px-4 py-6">
      <section className="mx-auto max-w-4xl">
        <Link
          href="/competitions"
          className="text-sm text-slate-500 underline"
        >
          â† Competitions
        </Link>

        <header className="mt-5 rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
          <div className="flex items-start justify-between gap-4">
            <div>
              <p className="text-xs font-semibold uppercase tracking-wide text-slate-500">
                Competition
              </p>

              <h1 className="mt-1 text-2xl font-bold text-slate-900">
                {competition.title}
              </h1>

              <p className="mt-2 text-sm text-slate-500">
                {formatPlayMode(
                  competition.play_mode
                )}{" "}
                Â·{" "}
                {formatStructureType(
                  competition.structure_type
                )}
              </p>
            </div>

            <span className="rounded-full bg-slate-100 px-3 py-1 text-xs font-semibold text-slate-600">
              {competition.status}
            </span>
          </div>
        </header>

        <nav className="sticky top-0 z-10 mt-5 overflow-x-auto border-y border-slate-200 bg-slate-50 py-3">
          <div className="flex min-w-max gap-2">
            {sections.map((section) => {
              const active =
                validSection === section.key

              return (
                <Link
                  key={section.key}
                  href={`/competitions/${competition.id}?section=${section.key}`}
                  className={`rounded-xl px-4 py-2 text-sm font-semibold ${
                    active
                      ? "bg-slate-900 text-white"
                      : "border border-slate-300 bg-white text-slate-700"
                  }`}
                >
                  {section.label}
                </Link>
              )
            })}
          </div>
        </nav>

        <section className="mt-5">
          {validSection === "entries" ? (
            <CompetitionEntriesManager
              competitionId={competition.id}
              entries={entries}
              locked={locked}
            />
          ) : (
            <div className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
              <h2 className="font-semibold text-slate-900">
                {sections.find(
                  (section) =>
                    section.key === validSection
                )?.label ?? "Configuration"}
              </h2>

              <p className="mt-1 text-sm text-slate-500">
                This section will be implemented
                in the next sprint.
              </p>
            </div>
          )}
        </section>
      </section>
    </main>
  )
}
