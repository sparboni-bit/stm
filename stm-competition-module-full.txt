"use server"

import { redirect } from "next/navigation"
import { getCurrentMember } from "@/lib/auth/getCurrentMember"
import { createCompetition } from "../repositories/competition.repository"
import type { PlayMode, StructureType } from "../types"

function getDefaultSettings() {
  return {
    scoring: {
      default_mode: "single_set",
    },
    courts: [],
  }
}

function getDefaultStructure(structureType: StructureType) {
  return {
    type: structureType,
    has_groups:
      structureType === "round_robin" ||
      structureType === "round_robin_bracket" ||
      structureType === "round_robin_bracket_consolation",
    has_main_bracket:
      structureType === "single_elimination" ||
      structureType === "round_robin_bracket" ||
      structureType === "round_robin_bracket_consolation",
    has_consolation: structureType === "round_robin_bracket_consolation",
  }
}

export async function createCompetitionAction(formData: FormData) {
  const title = String(formData.get("title") || "").trim()
  const playMode = String(formData.get("playMode") || "") as PlayMode
  const structureType = String(
    formData.get("structureType") || ""
  ) as StructureType

  if (!title || !playMode || !structureType) {
    redirect("/competitions/new?error=missing_fields")
  }

  const member = await getCurrentMember()

  if (!member) {
    redirect("/login?error=no_active_member")
  }

  const { data, error } = await createCompetition({
    organization_id: member.organization_id,
    owner_member_id: member.id,
    created_by: member.id,
    title,
    play_mode: playMode,
    structure_type: structureType,
    settings: getDefaultSettings(),
    structure: getDefaultStructure(structureType),
  })

  if (error || !data) {
    redirect("/competitions/new?error=create_failed")
  }

  redirect(`/competitions/${data.id}`)
}

"use server"

import { createClient } from "@/lib/supabase/server"
import { getCurrentMember } from "@/lib/auth/getCurrentMember"
import type { Competition, CompetitionListResult } from "../types"

function canSeeAllCompetitions(role: string) {
  return role === "owner" || role === "manager" || role === "viewer"
}

export async function listCompetitionsAction(): Promise<CompetitionListResult> {
  const member = await getCurrentMember()

  if (!member) {
    return {
      active: [],
      archived: [],
    }
  }

  const supabase = await createClient()

  let query = supabase
    .from("competitions")
    .select("*")
    .eq("organization_id", member.organization_id)
    .order("created_at", { ascending: false })

  if (!canSeeAllCompetitions(member.role)) {
    query = query.eq("owner_member_id", member.id)
  }

  const { data, error } = await query

  if (error || !data) {
    return {
      active: [],
      archived: [],
    }
  }

  const competitions = data as Competition[]

  return {
    active: competitions.filter((competition) => competition.status !== "archived"),
    archived: competitions.filter(
      (competition) => competition.status === "archived"
    ),
  }
}

import { createClient } from "@/lib/supabase/server"
import type { Competition } from "../types"

export async function createCompetition(data: {
  organization_id: string
  owner_member_id: string
  created_by: string
  title: string
  play_mode: string
  structure_type: string
  settings: object
  structure: object
}) {
  const supabase = await createClient()

  return supabase
    .from("competitions")
    .insert({
      ...data,
      status: "draft",
      metadata: {},
    })
    .select("id")
    .single()
}

export async function getCompetition(id: string) {
  const supabase = await createClient()

  return supabase
    .from("competitions")
    .select("*")
    .eq("id", id)
    .single<Competition>()
}

export async function listCompetitions(organizationId: string) {
  const supabase = await createClient()

  return supabase
    .from("competitions")
    .select("*")
    .eq("organization_id", organizationId)
    .order("created_at", { ascending: false })
}

export function validateCompetition(data: {
  title: string
  playMode: string
  structureType: string
}) {
  const errors: string[] = []

  if (!data.title.trim()) {
    errors.push("Title is required")
  }

  if (!data.playMode) {
    errors.push("Play mode is required")
  }

  if (!data.structureType) {
    errors.push("Structure type is required")
  }

  return errors
}
